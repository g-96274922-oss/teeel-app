// Set your desired Teacher PIN here const TEACHER_PIN = "1234"; function toggleMode(mode) { if (mode === 'teacher') { // If switching to Teacher Mode, prompt for PIN const enteredPin = prompt("Enter Teacher Security PIN to access grading & logs:"); if (enteredPin !== TEACHER_PIN) { alert("Incorrect PIN! Access denied."); return; // Stop execution, stay in Student Mode } } // Update UI mode buttons and active view document.getElementById('btn-student').classList.toggle('active', mode === 'student'); document.getElementById('btn-teacher').classList.toggle('active', mode === 'teacher'); document.getElementById('student-view').classList.toggle('hidden', mode !== 'student'); document.getElementById('teacher-view').classList.toggle('hidden', mode !== 'teacher'); }
